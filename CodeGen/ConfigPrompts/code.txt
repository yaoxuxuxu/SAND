//database.sand
class ProductDB {
    records = []
    next_id = 1

    def add(name, price, stock) {
        id = next_id
        next_id = next_id + 1
        records.append([id, name, price, stock])
        id
    }

    def find(id) {
        i = 0
        found = -1
        while (i < records.length()) {
            row = records[i]
            if (row[0] == id) { found = i }
            i = i + 1
        }
        if (found == -1) { -1 }
        else {
            row   = records[found]
            id    = row[0]
            name  = row[1]
            price = row[2]
            stock = row[3]
            return_all
        }
    }

    def updateStock(id, delta) {
        i = 0
        while (i < records.length()) {
            row = records[i]
            if (row[0] == id) {
                records[i][3] = row[3] + delta
            }
            i = i + 1
        }
    }

    def remove(id) {
        i = 0
        while (i < records.length()) {
            if (records[i][0] == id) {
                records.pop(i)
            }
            i = i + 1
        }
    }

    def stats() {
        count       = records.length()
        total_stock = 0
        best_id     = -1
        best_qty    = -1
        i = 0
        while (i < count) {
            row = records[i]
            total_stock = total_stock + row[3]
            if (row[3] > best_qty) {
                best_qty = row[3]
                best_id  = row[0]
            }
            i = i + 1
        }
        return_all
    }

    def all() {
        snap = []
        i = 0
        while (i < records.length()) {
            snap.append(records[i])
            i = i + 1
        }
        snap
    }
}

//console_ui.sand
class Formatter {
    width = 40

    def repeat(ch, n) {
        s = ""
        i = 0
        while (i < n) {
            s = s + ch
            i = i + 1
        }
        s
    }

    def line() {
        print(this.repeat("-", width))
    }

    def header(title) {
        this.line()
        print("| " + title)
        this.line()
    }
}

class ConsoleUI extends Formatter {

    def printRow(row) {
        id    = row[0]
        name  = row[1]
        price = row[2]
        stock = row[3]
        print("[" + str(id) + "] " + name +
              "  price:" + str(price) +
              "  stock:" + str(stock))
    }

    def listAll(rows) {
        this.header("Product List")
        i = 0
        while (i < rows.length()) {
            this.printRow(rows[i])
            i = i + 1
        }
        this.line()
    }

    def printItem(item) {
        if (item == -1) {
            print("[not found]")
        } else {
            print("Found: [" + str(item.id) + "] " + item.name +
                  "  price:" + str(item.price) +
                  "  stock:" + str(item.stock))
        }
    }

    def printStats(s) {
        this.header("Statistics")
        print("total products : " + str(s.count))
        print("total stock    : " + str(s.total_stock))
        print("most stocked id: " + str(s.best_id))
        this.line()
    }

    def makePrinter(prefix) {
        lambda(msg){ print(prefix + msg) }
    }
}

//main.sand
from database import ProductDB
from console_ui import ConsoleUI

db = ProductDB.new
ui = ConsoleUI.new

// ── 录入商品 ──────────────────────────────
db.add("Apple",  3,  120)
db.add("Banana", 1,  200)
db.add("Cherry", 15, 40)
db.add("Durian", 80, 8)

// ── 列出全部 ──────────────────────────────
ui.listAll(db.all())

// ── 查找 ──────────────────────────────────
item = db.find(2)
ui.printItem(item)

item = db.find(99)
ui.printItem(item)

// ── 更新库存 ──────────────────────────────
db.updateStock(1, -30)   // Apple 卖出 30
db.updateStock(3, 60)    // Cherry 补货 60

// ── 删除商品 ──────────────────────────────
db.remove(4)             // 下架 Durian

// ── 再次列出 ──────────────────────────────
ui.listAll(db.all())

// ── 统计 ──────────────────────────────────
s = db.stats()
ui.printStats(s)

// ── 闭包演示 ──────────────────────────────
warn = ui.makePrinter("[WARN] ")
info = ui.makePrinter("[INFO] ")
warn("Apple stock is low!")
info("System ready.")
