def function(){
    // Read the input integer n
    n = int(input());

    // Handle base cases for Fibonacci sequence
    // f(0) = 0
    if (n == 0) {
        print(0);
        return;
    }
    // f(1) = 1
    if (n == 1) {
        print(1);
        return;
    }

    // For n > 1, use an iterative approach
    // Initialize the first two Fibonacci numbers
    // f0 represents F(i-2)
    // f1 represents F(i-1)
    f0 = 0;
    f1 = 1;
    
    // Loop from i = 2 up to n to calculate F(n)
    // In each iteration, f_current = f0 + f1
    // Then update f0 to f1, and f1 to f_current
    for (i = 2; i <= n; i = i + 1) {
        next_fib = f0 + f1;
        f0 = f1;
        f1 = next_fib;
    }

    // After the loop, f1 will hold the value of F(n)
    print(f1);
}
