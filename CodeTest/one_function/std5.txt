def function(n){
    status=special(n)
    flag=true
    if status=="not special"{
        i=3
        while(i*i<=n){
            if(n%i==0){
                flag=false
            }
            i=i+2
        }
        flag
    }
    else{
        status
    }
    
}
def special(n){
    if n<2{
        false
    }
    else{
        if(n==2){
            true
        }
        else{
            if(n%2==0){
                false
            }
            else{
                "not special"
            }
        }
    }
}
